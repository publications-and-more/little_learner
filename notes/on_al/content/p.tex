\begin{theorem}
	There exist for $p, \operatorname{L}, f(a, qk), a, q, k \in \mathbb{R}.$:
	\begin{equation}
		\lim_{p \to \operatorname{L}} (\frac{f(a, qk)}{k} + \Bigl(\pi^{\frac{1}{12}} - \frac{\zeta(3)}{\pi}\Bigr))^p, \, f(a, qk), p \neq 0.
	\end{equation}
	Solutions of $\lim_{p \to \operatorname{L}} \Bigl(\frac{f(a, qk)}{k} + \Bigl(\pi^{\frac{1}{12}} - \frac{\zeta(3)}{\pi}\Bigr)\Bigr)^p, \, \operatorname{L} \in \{2k : k \in \mathbb{Z}\}.$
\end{theorem}
\begin{remark}
	There exist an infinite set of solutions in $\operatorname{L} \in \{2k : k \in \mathbb{Z}\}$ in parity.
\end{remark}
\begin{remark}
There exist an infinite set of solutions of transcendental property.
\end{remark}

\begin{remark}
	Per recent exploration of previously stated properties, It may be worth sharing this specific bounded integral:
	\begin{align*}
		\int_{\frac{1}{2}}^{2} \pi^{\frac{3}{12}} + \frac{\zeta(2)}{\pi} &= \frac{1}{4} \Bigl(6 \cdot \pi^{\frac{1}{4}} + \pi\Bigr).
	\end{align*}
	Admits a sequence of solutions in $\mathbb{R}$, of algebraic and transcendence property. Posing the limit of $(\pi^{\frac{3}{12}} + \frac{\zeta(2)}{\pi})$ in $+\infty$, we get:
	\begin{align*}
	\lim_{x \to 0} \Bigl(\pi^{\frac{3}{12}} + \frac{\zeta(2)}{\pi}\Bigr) &= \frac{1}{4} \Bigl(6 \cdot \pi^{\frac{1}{4}} + \pi\Bigr).
	\end{align*}
	Which seems to be generalizable under the following term for $n \neq 0, \, n \in \mathbb{N}.$ And a non-zero $p \in \mathbb{N}.$:
	\begin{align*}
		\int_{\frac{1}{p}}^{p} \pi^{\frac{3}{12}} + \frac{\zeta(n)}{\pi}.
	\end{align*}
\end{remark}

\begin{remark}
	Developping further from Remark 2.4, we establish the inequality consisting of the Riemann zeta function on the left:
	\begin{align*}
		\zeta\Bigl(\frac{1}{n}\Bigr) < \frac{1}{4}\Bigl(6 \cdot \pi^{\frac{1}{4}} + \pi\Bigr) - \zeta(3)
	\end{align*}

\end{remark}

\begin{corollary}
	There exist an indetity of the following for $\forall t, k, x \in \mathbb{R}$.:
	\begin{align*}
		\mathcal{M}(x, k) = \int_{n}^{p} \mathcal{M}_{x}(\mathcal{X}_k) \cdot t \, dt,\\
		\mathcal{M}(x, k)^{2} = \int_{n}^{p} \mathcal{M}_{x}(\mathcal{X}_k) \cdot t \, dt^{2}.
	\end{align*}
	And thus we may write the following:
	\begin{align*}
		\mathcal{M}(x, k)^{2} + \mathcal{M}(x, k).
	\end{align*}
	Which then could be rewritten as:
	\begin{align*}
		F(x, k) &= \mathcal{M}(x, k)^{2} + \mathcal{M}(x, k).
	\end{align*}
	Adding the tensors defined as: $\mathcal{T}_3, \mathcal{T}_2 \in \mathbb{R}.$:
	\begin{align*}
		F(x, k) + \mathcal{T}_2 &= \mathcal{M}(x, k)^{2} + \mathcal{M}(x, k) + \mathcal{T}_3.
	\end{align*}
	A more complete definition with $\mathcal{T}_2$ and $\mathcal{T}_3$ would then be:
	\begin{align*}
		\mathcal{T}_2 &= \mathcal{M}(x, k)^{2} + \mathcal{M}(x, k)^{1} + \mathcal{T}_3 - F(x, k), \, \forall x, k \in \mathbb{R}.
	\end{align*}
	A more rigorous form of the equation above would be:
	\begin{align*}
			\pm \mathcal{T}_2 &= \mathcal{M}(x, k)^{2} + \mathcal{M}(x, k)^{1} + \mathcal{T}_3 \pm F(x, k), \, \forall x, k \in \mathbb{R}.
	\end{align*}
\end{corollary}
