\section{Observations:}

\begin{definition}
	We define the following identity below in $\mathbb{C}.$ Such that:
	\begin{align*}
		\ln(-1)&=\ln(e^{i \cdot \pi}), \, i \in \mathbb{C}.
	\end{align*}
	Thus:
	\begin{center}
		\fbox{$\ln(-1)=\ln(e^{i \cdot \pi}), \, i \in \mathbb{C}.$}
	\end{center}
	Let us define the Lambert W function, named after Jean-Henri-Lambert, for $x \in \mathbb{C}.$ Such that:
	\begin{align*}
		\exists W(z) \in \mathbb{C},\\
		f(z) = z \cdot e^{z}, \, f(z) &= W(z).
	\end{align*}
\end{definition}

\begin{corollary}
	Let us define the following relation for $z, z' \in \mathbb{C},$ which states:
	\begin{align*}
		W(z') &= W(z), \, z', z \in \mathbb{C}.
	\end{align*}
\end{corollary}

\begin{proof}
	Per corollary 2.2, $\ln(-1)=\ln(e^{i \cdot \pi})$ can be expressed by construction in the form of:
	\begin{equation}
		W(\ln(-1)) = W(\ln(e^{i \cdot \pi}))
	\end{equation}
	(i) By the following identity of $\ln(-1)=\ln(e^{i \cdot \pi})$ and $-1 = e^{i \cdot \pi}.$\\
	(ii) And that the identity of $\ln(-1) = \pi \cdot i$, by construction yields the identity $\ln(-1) = \ln(e^{i \cdot \pi}).$\\
	(iii) Thus proving by construction the following identity for $e^{i \cdot \pi}$. This completes the proof.
\end{proof}

\begin{lemma}
	Defining the following sequence $\mathcal{S}_k$ for $k \in \mathbb{N}$, we have:
	\begin{equation}
		\mathcal{S}_k = \sqrt[p]{k}
	\end{equation}
	And thus we may write the next sequence $\mathcal{S}_{k+1}$:
	\begin{equation}
		\mathcal{S}_{k+1} = \sqrt[p]{k+1}
	\end{equation}
	Thus we have a difference variable $n \in \mathbb{N}$, such that:
	\begin{equation}
		\mathcal{S}_{k} - \mathcal{S}_{k+1} = \{ n \}
	\end{equation}
	For our discrete sequence $\mathcal{S}_k$ in $\mathbb{R}$.
\end{lemma}
